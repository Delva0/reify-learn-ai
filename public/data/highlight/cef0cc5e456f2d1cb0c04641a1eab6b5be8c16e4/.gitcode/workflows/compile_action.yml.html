name: compile_action

on:
  workflow_call:
    inputs:
      task_name:
        required: true
        type: string
      image_version:
        required: true
        type: string
      arch:
        required: true
        type: string
    outputs:
      arm_run_url:
        value: ${{jobs.JOB_Compile.outputs.primary-url}}
      arm_run_key:
        value: ${{jobs.JOB_Compile.outputs.primary-key}}

jobs:
  JOB_Compile:
    name: Compile
    steps:
      -
        name: Checkout
        identifier: process_checkout
        uses: checkout
        with:
          ref: ${{ atomgit.event.pull_request.merge_commit_sha }}
          allow-unsafe-pr-checkout: true
      -
        name: download
        uses: obs-download
        with:
          endpoint: "https://obs.cn-north-4.myhuaweicloud.com"
          bucket: "ascend-ci"
          access-key: ${{ secrets.AK }}
          secret-key: ${{ secrets.SK }}
          key: |
            ${{ env.obs_path }}/pr_filelist.txt
            ${{ env.obs_path }}/pr_filelist_mod.txt
            ${{ env.obs_path }}/update_file_detail.txt
            ${{ env.obs_path }}/triton-inference-server-ge-backend_compile.sh
          path: ${{ steps.process_checkout.outputs.path }}
      -
        name: compile_acc
        identifier: compile
        uses: build-accelerate
        with:
          command: |
            export WORKSPACE=${{ steps.process_checkout.outputs.path }}
            cd "${{ steps.process_checkout.outputs.path }}" || exit 1
            export task_name=${{ inputs.task_name }}
            export GIT_TARGET_BRANCH=${{ env.TARGET_BRANCH }}
            export ASCEND_3RD_LIB_PATH=/home/jenkins/opensource
            BuildAccelerate bash triton-inference-server-ge-backend_compile.sh
          AC_SERVER_IP: ${{ vars.AC_SERVER_IP }}
      - name: upload
        uses: obs-upload
        with:
          endpoint: "https://obs.cn-north-4.myhuaweicloud.com"
          bucket: "ascend-ci"
          access-key: ${{ secrets.AK }}
          secret-key: ${{ secrets.SK }}
          artifact-path: |
            ${{ steps.process_checkout.outputs.path }}/output/*.tar
            ${{ steps.process_checkout.outputs.path }}/tf_adaoter_2.x/build/dist/python/dist/*.whl
          object-prefix: ${{ env.obs_path }}/
    if: "${{ default() }}"
    runs-on: ['self-hosted', "arch=${{ inputs.arch }}", 'spec=xlarge']
    container:
      image: swr.cn-north-4.myhuaweicloud.com/ci_cann/${{ inputs.image_version }}
